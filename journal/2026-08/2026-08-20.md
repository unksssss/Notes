继承 `WristStepPanel<EquipDefinition, VerticalBackPackItem>`，对接消防车装备确认链路。今日改动集中在**防御性边界防护**：

```cs
public override void Start()
{
    base.Start();
    if (fireTruckUI != null)
    {
        // 防重复订阅：先取消再加，避免 Start 多次调用导致事件叠加
        fireTruckUI.OnConfirmEquips -= AddEquip;
        fireTruckUI.OnConfirmEquips += AddEquip;
    }
    else Debug.LogError("未添加消防车工具选择面板引用");
}

public override void Init()
{
    ...
    // itemList 包含 rightHand 默认项 + dataList 装备，用实际数量算最大索引
    // 边界防护：列表为空时 maxIndex=0，Highlight 不越界
    maxIndex = (itemList != null && itemList.Count > 0) ? itemList.Count - 1 : 0;
    Highlight(Mathf.Min(0, maxIndex));
}

public override void Confirm()
{
    if (itemList == null || itemList.Count == 0) return;   // 空列表防护
    index = Mathf.Clamp(index, 0, itemList.Count - 1);     // 越界钳制
    var chooseEquip = itemList[index]?.equip;
    if (chooseEquip == null) return;
    onChooseData?.Invoke(chooseEquip);
    ...
    if (itemDict.TryGetValue(chooseEquip, out var targetItem) && targetItem != null)
        targetItem.img_Status.sprite = sprite_Choose;      // 选中图标防崩
    if (EquipmentManager.Instance != null)
        EquipmentManager.Instance.Equip(chooseEquip);     // 装备链路
    if (TipsPopup.instance != null)
        TipsPopup.instance.Show($"装备了 {chooseEquip.displayName}");
}

protected override void Highlight(int _index)
{
    if (itemList == null || itemList.Count == 0) return;   // 空列表不崩溃
    _index = Mathf.Clamp(_index, 0, itemList.Count - 1);   // 越界钳制
    itemList[_index].toggle.isOn = true;
}
```

要点：
- **`-=` 再 `+=` 的防重复订阅模式**：Unity 事件多次 `+=` 会累积委托链，`Start` 若被重复触发（热重载/多实例）会叠加回调；先取消再加保证幂等。
- **空列表三处兜底**：`Init` 的 maxIndex、`Confirm` 的空列表 return + Clamp、`Highlight` 的空列表 return + Clamp——手腕背包在装备未确认前 itemList 可能为空，任何一处裸索引都会越界崩溃。
- 装备计数：`tmp_EquipCount.text = $"{_index}"` 显示"右手默认项 + dataList 装备"总数。
- 旧链路保留：`AddEquip` 清空重建（先 `SetActive(false)` 再 `Destroy`，防 `stepScrollView.CollectItems` 只收 active 子物体时收集到旧 item 导致计数错乱）、`itemDict.ContainsKey` 防同装备重复 key 抛异常。

```cpp
struct TreeNode {
    int val;
    TreeNode* left;
    TreeNode* right;
    TreeNode(int v) : val(v), left(nullptr), right(nullptr) {}
};

void PreOrder(TreeNode* root)   { if (!root) return; visit(root); PreOrder(root->left);  PreOrder(root->right); }
void InOrder(TreeNode* root)    { if (!root) return; InOrder(root->left); visit(root);   InOrder(root->right); }
void PostOrder(TreeNode* root)  { if (!root) return; PostOrder(root->left); PostOrder(root->right); visit(root); }
```
